{{/* gm-img — image for print: a single JPEG large enough for ~300 dpi at A4 width, instead of the web partial's WebP srcset. Params: src (assets path), alt, class, style. */}} {{- with resources.Get (strings.TrimPrefix "/" .src) -}} {{- $img := . -}} {{- if eq $img.MediaType.SubType "png" }}{{/* cut-out photos: keep the transparency */}} {{- else if gt $img.Width 2200 }}{{ $img = $img.Process "resize 2200x jpg q85" }}{{ else }}{{ $img = $img.Process "jpg q90" }}{{ end -}} {{- else -}} {{- errorf "gm-img: image %q not found" .src -}} {{- end -}}
